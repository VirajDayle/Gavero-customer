// --- FLAT DATA BUILDER ---
// Pure function to flatten sections & sub-sections into a simple list
function buildFlatData(sections: MenuSection[], openSubsections: Set<string>): FlatRow[] {
  const result: FlatRow[] = [
    { type: "headerSpacer", id: "header-spacer", sectionIndex: -1 },
    { type: "sticky-header", id: "sticky-header", sectionIndex: -1 },
  ];

  sections.forEach((section, sectionIdx) => {
    let count = section.items.length;
    section.subsections.forEach((sub) => (count += sub.items.length));

    result.push({
      type: "section-header",
      id: `sec-${section.id}`,
      title: section.title,
      count,
      sectionIndex: sectionIdx,
    });

    section.items.forEach((item) => {
      result.push({ type: "item", id: `item-${item.id}`, item, sectionIndex: sectionIdx });
    });

    section.subsections.forEach((sub) => {
      const isOpen = openSubsections.has(sub.id);
      result.push({
        type: "subsection-toggle",
        id: `sub-${sub.id}`,
        title: sub.title,
        isOpen,
        sectionId: sub.id,
        sectionIndex: sectionIdx,
      });

      if (isOpen) {
        sub.items.forEach((item) => {
          result.push({ type: "subsection-item", id: `subitem-${item.id}`, item, sectionIndex: sectionIdx });
        });
      }
    });
  });

  return result;
}

    });

    return (

const AddToCartButton = () => {
  const [count, setCount] = useState(0);

  if (count === 0) {
    return (
      <TouchableOpacity
        onPress={() => setCount(1)}
        className="bg-white border border-[#FF6347] rounded-md px-6 py-1.5 shadow-sm"
      >
        <Text className="text-[#FF6347] font-bold text-center">ADD</Text>
      </TouchableOpacity>
    );
  }

  return (
    <Animated.View
      entering={FadeIn}
      exiting={FadeOut}
      className="flex-row items-center bg-[#FF6347] rounded-md shadow-sm overflow-hidden"
    >
      <TouchableOpacity onPress={() => setCount((c) => c - 1)} className="px-3 py-1.5">
        <Text className="text-white font-bold text-lg leading-5">-</Text>
      </TouchableOpacity>
      <Text className="text-white font-bold px-1">{count}</Text>
      <TouchableOpacity onPress={() => setCount((c) => c + 1)} className="px-3 py-1.5">
        <Text className="text-white font-bold text-lg leading-5">+</Text>
      </TouchableOpacity>
    </Animated.View>
  );
};

interface RestaurantShopProps {
  sections: Sections; // From parent, unused in mock
  onScroll: ReturnType<typeof useAnimatedScrollHandler>;
  headerHeightSv: SharedValue<number>;
  currentShop: ShopType;
  changeShop: React.Dispatch<React.SetStateAction<ShopType>>;
  scrollY: SharedValue<number>;
}

const RestaurantShop = ({
  onScroll,
  headerHeightSv,
  currentShop,
  changeShop,
  scrollY,
}: RestaurantShopProps) => {
  const scrollThresholdSv = useSharedValue(200);
  const flashListRef = useRef<FlashList<FlatRow>>(null);
  const chipListRef = useRef<FlashList<MenuSection>>(null);
  const bottomSheetModalRef = useRef<BottomSheetModal>(null);

  const [activeIndex, setActiveIndex] = useState(0);
  const [openSubsections, setOpenSubsections] = useState<Set<string>>(new Set());

  // Use TanStack Query to fetch mock data
  const { data: menuData = [] } = useQuery({
    queryKey: ["restaurantMenu"],
    queryFn: () => Promise.resolve(MOCK_DATA),
  });

  const extendedData = useMemo(() => buildFlatData(menuData, openSubsections), [menuData, openSubsections]);

  const headerSpacerStyle = useAnimatedStyle(() => ({
    height: headerHeightSv.value,
  }));

  const onChipPress = useCallback(
    (index: number) => {
      setActiveIndex(index);
      try {
        chipListRef.current?.scrollToIndex({ index, animated: true, viewPosition: 0.5 });
      } catch (e) {}

      // Find the first header of the target section and scroll there
      const targetRowIndex = extendedData.findIndex(
        (row) => row.type === "section-header" && row.sectionIndex === index
      );
      if (targetRowIndex !== -1) {
        flashListRef.current?.scrollToIndex({ index: targetRowIndex, animated: true, viewOffset: 120 });
      }
    },
    [extendedData]
  );

  const onToggleSubsection = useCallback((sectionId: string) => {
    LayoutAnimation.configureNext(LayoutAnimation.Presets.easeInEaseOut);
    setOpenSubsections((prev) => {
      const next = new Set(prev);
      if (next.has(sectionId)) next.delete(sectionId);
      else next.add(sectionId);
      return next;
    });
  }, []);

  const viewabilityConfig = useRef({ itemVisiblePercentThreshold: 30 }).current;

  const onViewableItemsChanged = useRef(({ viewableItems }: { viewableItems: ViewToken[] }) => {
    // Find the first item that actually belongs to a section
    const visibleItem = viewableItems.find((v) => v.item.sectionIndex !== undefined && v.item.sectionIndex >= 0);
    if (visibleItem) {
      const idx = visibleItem.item.sectionIndex;
      if (idx !== undefined) {
        setActiveIndex((prevIdx) => {
          if (prevIdx !== idx) {
            try {
              chipListRef.current?.scrollToIndex({ index: idx, animated: true, viewPosition: 0.5 });
            } catch (e) {}
            return idx;
          }
          return prevIdx;
        });
      }
    }
  }).current;

  const openMenu = useCallback(() => {
    bottomSheetModalRef.current?.present();
  }, []);

  const closeMenu = useCallback(() => {
    bottomSheetModalRef.current?.dismiss();
  }, []);

  const renderBackdrop = useCallback(
    (props: any) => (
      <BottomSheetBackdrop {...props} disappearsOnIndex={-1} appearsOnIndex={0} opacity={0.4} />
    ),
    []
  );

  const onMenuSelect = useCallback(
    (sectionIndex: number, subsectionId?: string) => {
      closeMenu();

      if (subsectionId) {
        setOpenSubsections((prev) => {
          if (!prev.has(subsectionId)) {
            LayoutAnimation.configureNext(LayoutAnimation.Presets.easeInEaseOut);
            const next = new Set(prev);
            next.add(subsectionId);
            return next;
          }
          return prev;
        });
      }

      setActiveIndex(sectionIndex);
      try {
        chipListRef.current?.scrollToIndex({ index: sectionIndex, animated: true, viewPosition: 0.5 });
      } catch (e) {}

      // Allow state update to propagate
      setTimeout(() => {
        // Because of closure, we might not have the newest extendedData, but the headers and sub-headers
        // toggle rows are ALWAYS physically in the array regardless of subsection open state.
        const targetRowIndex = extendedData.findIndex((row) => {
          if (subsectionId) {
            return row.type === "subsection-toggle" && (row as any).sectionId === subsectionId;
          } else {
            return row.type === "section-header" && row.sectionIndex === sectionIndex;
          }
        });

        if (targetRowIndex !== -1) {
          flashListRef.current?.scrollToIndex({ index: targetRowIndex, animated: true, viewOffset: 120 });
        }
      }, 100);
    },
    [closeMenu, extendedData]
  );

  const renderItem = useCallback(
    ({ item }: { item: FlatRow }) => {
      if (item.type === "headerSpacer") {
        return (
          <HeaderSpacerItem
            style={headerSpacerStyle}
            currentShop={currentShop}
            changeShop={changeShop}
            onLayout={(e) => {
              scrollThresholdSv.value = e.nativeEvent.layout.height;
            }}
          />
        );
      }

      if (item.type === "sticky-header") {
        return (
          <StickyHeaderItem
            scrollY={scrollY}
            scrollThresholdSv={scrollThresholdSv}
            sections={menuData}
            activeIndex={activeIndex}
            onChipPress={onChipPress}
            chipListRef={chipListRef}
            onMenuPress={openMenu}
          />
        );
      }

      if (item.type === "section-header") {
        return <SectionHeaderRow title={item.title} count={item.count} />;
      }

      if (item.type === "subsection-toggle") {
        return (
          <SubsectionToggleRow
            title={item.title}
            isOpen={item.isOpen}
            onToggle={() => onToggleSubsection(item.sectionId)}
          />
        );
      }

      if (item.type === "item") {
        return <MenuItemRow item={item.item} />;
      }

      if (item.type === "subsection-item") {
        return <MenuItemRow item={item.item} isSubsection />;
      }

      return null;
    },
    [
      headerSpacerStyle,
      currentShop,
      changeShop,
      scrollY,
      scrollThresholdSv,
      menuData,
      activeIndex,
      onChipPress,
      onToggleSubsection,
      openMenu,
    ]
  );

  const snapPoints = useMemo(() => ["50%", "80%"], []);

  return (
    <View className="flex-1 z-10 bg-white">
      <AnimatedFlashList
        ref={flashListRef as any}
        data={extendedData}
        keyExtractor={(item: any) => item.id}
        renderItem={renderItem as any}
        getItemType={(item: any) => item.type}
        stickyHeaderIndices={[1]}
        contentContainerStyle={{ paddingBottom: 100 }}
        showsVerticalScrollIndicator={false}
        onScroll={onScroll}
        scrollEventThrottle={16}
        estimatedItemSize={120}
        viewabilityConfig={viewabilityConfig}
        onViewableItemsChanged={onViewableItemsChanged}
      />

      <BottomSheetModal
        ref={bottomSheetModalRef}
        index={0}
        snapPoints={snapPoints}
        backdropComponent={renderBackdrop}
      >
        <View className="flex-1 bg-white">
          <View className="px-5 py-4 border-b border-gray-200 flex-row justify-between items-center">
            <Text className="text-xl font-bold text-gray-900">Menu</Text>
            <TouchableOpacity onPress={closeMenu} className="p-1">
              <Ionicons name="close" size={24} color="#6B7280" />
            </TouchableOpacity>
          </View>
          <BottomSheetScrollView contentContainerStyle={{ paddingBottom: 30 }}>
            {menuData.map((section, sIdx) => {
              let secTotalCount = section.items.length;
              section.subsections.forEach((sub) => (secTotalCount += sub.items.length));

              return (
                <View key={section.id} className="border-b border-gray-100">
                  <TouchableOpacity
                    className="flex-row justify-between items-center px-5 py-4"
                    onPress={() => onMenuSelect(sIdx)}
                  >
                    <Text className="text-base font-bold text-gray-800">{section.title}</Text>
                    <Text className="text-sm font-semibold text-gray-500">{secTotalCount}</Text>
                  </TouchableOpacity>

                  {section.subsections.map((sub) => (
                    <TouchableOpacity
                      key={sub.id}
                      className="flex-row justify-between items-center pl-10 pr-5 py-3 bg-gray-50"
                      onPress={() => onMenuSelect(sIdx, sub.id)}
                    >
                      <Text className="text-sm font-medium text-gray-600">{sub.title}</Text>
                      <Text className="text-xs font-semibold text-gray-400">{sub.items.length}</Text>
                    </TouchableOpacity>
                  ))}
                </View>
              );
            })}
          </BottomSheetScrollView>
        </View>
      </BottomSheetModal>
    </View>
  );
};

export default RestaurantShop;
